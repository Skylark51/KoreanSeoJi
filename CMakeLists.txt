cmake_minimum_required(VERSION 3.24)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(KoreanSeoJi VERSION 0.1.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(sqlite STATIC .deps/sqlite/sqlite3.c)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0)
add_executable(KoreanSeoJi WIN32 src/main.cpp build/resources.rc)
target_include_directories(KoreanSeoJi PRIVATE .deps/webview/build/native/include .deps/json .deps/sqlite build)
target_compile_definitions(KoreanSeoJi PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(KoreanSeoJi PRIVATE /utf-8 /W4 /EHsc)
target_link_options(KoreanSeoJi PRIVATE /MANIFEST:NO /INCREMENTAL:NO)
target_link_libraries(KoreanSeoJi PRIVATE sqlite "${CMAKE_SOURCE_DIR}/.deps/webview/build/native/x64/WebView2LoaderStatic.lib" user32 ole32 oleaut32 shell32 shlwapi comdlg32 advapi32 version)
set_target_properties(KoreanSeoJi PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/dist")
